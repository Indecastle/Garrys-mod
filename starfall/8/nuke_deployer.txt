function randvec()
    local g = Vector(math.rand(-1,1),math.rand(-1,1),math.rand(-1,1))
    if g == Vector() then
        return Vector()
    else
        return g:getNormalized()
    end
end
function randang()

    function deploy(pos,ang,frozen)
        local bomb = wall(pos,Vector(28,28,106))
        bomb:setAngles(ang)
        bomb:setMass(100)
        bomb:setFrozen(frozen)
        net.start("bombify")
        net.writeEntity(bomb)
        net.send(find.allPlayers(),false)
        bomb:addCollisionListener(function(dat) if !bomb:isPlayerHolding() and (dat.HitEntity == game.getWorld() or dat.HitEntity:isFrozen()) and dat.HitNormal[3] < -0.15 and dat.HitSpeed:getLength() > 10^3 then if bomb:getWaterLevel() <= 1 then babaxx(bomb:getPos()) net.start("boom") net.writeVector(bomb:getPos()) net.send(find.allPlayers(),false) else net.start("waterboom") net.writeVector(bomb:getPos()) net.send(find.allPlayers(),false) end bomb:removeCollisionListener() bomb:remove() end end)
        function bomb:detonate()
            if bomb:getWaterLevel() <= 1 then babaxx(bomb:getPos()) net.start("boom") net.writeVector(bomb:getPos()) net.send(find.allPlayers(),false) else net.start("waterboom") net.writeVector(bomb:getPos()) net.send(find.allPlayers(),false) end bomb:removeCollisionListener() bomb:remove()
        end
        return bomb
    end

            local b = deploy(chip():getPos()+chip():getUp()*100+Vector(math.rand(-1,1),math.rand(-1,1),0):getNormalized()*100,Angle(),false)
        end
    end)
