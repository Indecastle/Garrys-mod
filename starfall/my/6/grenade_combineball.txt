local TriggerClasses = {
    'npc_grenade_frag',
    'grenade_ar2',
    'rpg_missile'
}

local _e = chip()

if SERVER then
    _e:setMaterial("models/debug/debugwhite")

    return
end


function table.first(t, func)    
    for i=1, #t do
        if func(t[i]) then
            return t[i]
        end
    end
    return nil
end



local eff = effect.create()

function start(pos)
    sound.emitSound("NPC_CombineBall.Explosion", pos, 75, 100, 1, 0, 1, false)
    
    eff:setOrigin(pos)
    eff:play("cball_explode")
    
    if render.isHUDActive() then
        --render.screenShake(20, 150, 1)
    end
    
    hook.add("RenderScene", "beamRingPoint1", function(origin, angles, fov)
        --effect.beamRingPoint(Vector pos, number lifetime, number startRad, number endRad, number width, number amplitude, Color color, number or nil speed, number or nil flags, number or nil framerate, string or nil material)
        effect.beamRingPoint(pos, 0.2, 12, 1024, 64, 0, Color(255,255,225,32), 0, nil, 2, "sprites/lgtning.vmt")
        hook.remove("RenderScene", "beamRingPoint1")
    end)
    
    hook.add("RenderScene", "beamRingPoint2", function(origin, angles, fov)
        --effect.beamRingPoint(Vector pos, number lifetime, number startRad, number endRad, number width, number amplitude, Color color, number or nil speed, number or nil flags, number or nil framerate, string or nil material)
        effect.beamRingPoint(pos, 0.5, 12, 1024, 64, 0, Color(255,255,225,64), 0, nil, 2, "sprites/lgtning.vmt")
        hook.remove("RenderScene", "beamRingPoint2")
    end)
    
end


hook.add('EntityRemoved', '', function(ent)
    if not ent or not isValid(ent) or not table.hasValue(TriggerClasses, ent:getClass()) then 
        --if isValid(ent) and ent:getPos():getDistance(eyePos()) < 1000 then
        --    print(ent:getClass()) 
        --end
    return end
    start(ent:getPos())
end)
