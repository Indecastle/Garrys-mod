local _screenSize = 512


local _maxQuota = SERVER and math.min(ServerQuota, quotaMax() * ServerQuotaP) or math.min(ClientQuota, quotaMax() * ClientQuotaP)

function checkQuota()
    return math.max(quotaAverage(), quotaUsed()) < _maxQuota
end

function validQuota()
    while not checkQuota() do
        coroutine.yield()
    end
end


local _locks = {}
function isLockEvent(name, sec)
    local curtime = timer.curtime()
    _locks[name] = _locks[name] or curtime
    local isLock = curtime < _locks[name]
    if sec ~= nil and not isLock then
        _locks[name] = curtime + sec
    end

    return isLock
end

function setLockEvent(name, sec)
    _locks[name] = timer.curtime() + sec
end

-- Shared drawing code (every client rasterizes operations the same way)
--
-- Operations (canvas pixels 0 .. 511):
--   L  line      X0 Y0 X1 Y1 Sz Rd C
--   R  rect      X Y W H C                  (filled)
--   B  rect line X Y W H Sz C               (outline)
--   E  ellipse   X Y W H Sz F Rd C          (bounding box, F = filled)
--   F  spans     Sp{y,x,w, y,x,w ..} C      (result of the bucket fill)

local function clipRect(x, y, w, h)
    if x < 0 then w = w + x; x = 0 end
    if y < 0 then h = h + y; y = 0 end
    if x + w > _screenSize then w = _screenSize - x end
    if y + h > _screenSize then h = _screenSize - y end
    if w <= 0 or h <= 0 then return nil end
    return x, y, w, h
end


-- Bresenham, calls plot() every `step` pixels (and always on the last pixel)
local function plotLine(x0, y0, x1, y1, step, plot)
    local dx = abs(x1 - x0)
    local dy = abs(y1 - y0)
    local sx = x0 < x1 and 1 or -1
    local sy = y0 < y1 and 1 or -1
    local err = dx - dy
    local n = 0

    while true do
        if n % step == 0 then plot(x0, y0) end

        if x0 == x1 and y0 == y1 then
            if n % step ~= 0 then plot(x0, y0) end
            break
        end

        local e2 = 2 * err
        if e2 > -dy then
            err = err - dy
            x0 = x0 + sx
        end
        if e2 < dx then
            err = err + dx
            y0 = y0 + sy
        end
        n = n + 1
    end
end


local _roundShapes = {}
local function getRoundShape(d)
    local shape = _roundShapes[d]
    if shape then return shape end

    shape = {}
    local r = d / 2
    for oy = 0, d - 1 do
        local yc = oy + 0.5 - r
        local hw = sqrt(max(r * r - yc * yc, 0))
        local x0 = floor(r - hw + 0.5)
        local x1 = floor(r + hw + 0.5)
        if x1 <= x0 then x1 = x0 + 1 end
        shape[#shape + 1] = x0
        shape[#shape + 1] = oy
        shape[#shape + 1] = x1 - x0
    end
    _roundShapes[d] = shape
    return shape
end


local function stamp(px, py, d, round, rect)
    if d <= 1 then
        rect(px, py, 1, 1)
        return
    end

    local half = floor(d / 2)
    local left, top = px - half, py - half
    if round and d > 2 then
        local shape = getRoundShape(d)
        for i = 1, #shape, 3 do
            rect(left + shape[i], top + shape[i + 1], shape[i + 2], 1)
        end
    else
        rect(left, top, d, d)
    end
end


local function stepFor(d, round)
    if d <= 2 then return 1 end
    if round then return max(1, floor(d / 6)) end
    return max(1, floor(d / 2))
end


local function runEllipse(op, rect)
    local x, y, w, h = op.X, op.Y, op.W, op.H
    if w < 1 or h < 1 then return end

    local rx, ry = w / 2, h / 2
    local cx, cy = x + rx, y + ry

    if op.F then
        for row = 0, h - 1 do
            local dy = (row + 0.5 - ry) / ry
            local t = 1 - dy * dy
            if t > 0 then
                local hw = rx * sqrt(t)
                local x0 = floor(cx - hw + 0.5)
                local x1 = floor(cx + hw + 0.5)
                if x1 <= x0 then x1 = x0 + 1 end
                rect(x0, y + row, x1 - x0, 1)
            end
        end
        return
    end

    local d, round = op.Sz, op.Rd
    local step = stepFor(d, round)
    local erx, ery = max(0.5, rx - d / 2), max(0.5, ry - d / 2)
    local n = math.clamp(floor(math.pi * (erx + ery)), 12, 400)
    local px, py
    for i = 0, n do
        local a = i / n * 2 * math.pi
        local sx = floor(cx + erx * math.cos(a))
        local sy = floor(cy + ery * math.sin(a))
        if px then
            plotLine(px, py, sx, sy, step, function(lx, ly) stamp(lx, ly, d, round, rect) end)
        else
            stamp(sx, sy, d, round, rect)
        end
        px, py = sx, sy
    end
end


-- rect(x, y, w, h) is a callback that fills a rectangle with the op color
local function runOp(op, rect)
    local t = op.T
    if t == "L" then
        local d, rd = op.Sz, op.Rd
        plotLine(op.X0, op.Y0, op.X1, op.Y1, stepFor(d, rd), function(x, y)
            stamp(x, y, d, rd, rect)
        end)
    elseif t == "R" then
        rect(op.X, op.Y, op.W, op.H)
    elseif t == "B" then
        local x, y, w, h, d = op.X, op.Y, op.W, op.H, op.Sz
        if w <= 2 * d or h <= 2 * d then
            rect(x, y, w, h)
        else
            rect(x, y, w, d)
            rect(x, y + h - d, w, d)
            rect(x, y + d, d, h - 2 * d)
            rect(x + w - d, y + d, d, h - 2 * d)
        end
    elseif t == "E" then
        runEllipse(op, rect)
    elseif t == "F" then
        local sp = op.Sp
        for i = 1, #sp - 2, 3 do
            rect(sp[i + 1], sp[i], sp[i + 2], 1)
        end
    end
end

    for i = #st.recent, 1, -1 do
        if sameColor(st.recent[i], c) then table.remove(st.recent, i) end
    end
    table.insert(st.recent, 1, c)
    while #st.recent > 8 do table.remove(st.recent) end
    st.dirty = true

local function mirrorOp(op, fx, fy)
    local M = _screenSize
    local n = {}
    for k, v in pairs(op) do n[k] = v end
    n.S = nil

    local function X(x) return fx and (M - 1 - x) or x end
    local function Y(y) return fy and (M - 1 - y) or y end

    if op.T == "L" then
        n.X0, n.X1 = X(op.X0), X(op.X1)
        n.Y0, n.Y1 = Y(op.Y0), Y(op.Y1)
    elseif op.T == "F" then
        -- fills are never mirrored
    else
        if fx then n.X = M - op.X - op.W end
        if fy then n.Y = M - op.Y - op.H end
    end
    return n
end

local function sendOp(op)
    _bufferNewData[#_bufferNewData + 1] = op
    drawOpLocal(op)
end

end

-- Reads the canvas (one row per capture) and returns it compressed.
-- Raw data: 3 bytes per pixel, rows 0..511, columns 0..511.
-- Formats  : "\1PC2" + chunks                -> full resolution, lossless
--            "\1PC3" + scale byte + chunks   -> reduced resolution (1/scale), 4 bit colors
--            (chunk = 16 rows: [4 byte length][bit.compress(rows)])
--            plain bit.compress(whole canvas) -> old format, still loads
local ROWS_PER_CHUNK = 16
local SCREEN_MAGIC = "\1PC2"
local SCREEN_MAGIC3 = "\1PC3"

local function encodeChunks(rowStrings, progressCallback, p0, p1)
    local out = {}
    local rows = {}
    local char = string.char

    local function flushChunk()
        if #rows == 0 then return end
        validQuota()
        local chunk = bit.compress(table.concat(rows))
        rows = {}
        local len = #chunk
        out[#out + 1] = char(floor(len / 16777216) % 256, floor(len / 65536) % 256, floor(len / 256) % 256, len % 256)
        out[#out + 1] = chunk
        validQuota()
    end

    for i = 1, #rowStrings do
        rows[#rows + 1] = rowStrings[i]
        if #rows >= ROWS_PER_CHUNK then
            flushChunk()
            progressCallback(p0 + (p1 - p0) * i / #rowStrings)
        end
    end
    flushChunk()

    return table.concat(out)
end


-- box average of scale x scale pixels, colors reduced to 4 bits
local function downscaleRows(rowStrings, scale, srcSize)
    local byte, char = string.byte, string.char
    local size = floor((srcSize or _screenSize) / scale)
    local n = scale * scale
    local result = {}

    for oy = 0, size - 1 do
        local srcRows = {}
        for dy = 1, scale do
            srcRows[dy] = rowStrings[oy * scale + dy]
        end

        local row = {}
        for ox = 0, size - 1 do
            local sr, sg, sb = 0, 0, 0
            local idx = ox * scale * 3 + 1
            for dy = 1, scale do
                local src = srcRows[dy]
                for dx = 0, scale - 1 do
                    local r, g, b = byte(src, idx + dx * 3, idx + dx * 3 + 2)
                    sr = sr + r
                    sg = sg + g
                    sb = sb + b
                end
            end
            row[ox + 1] = char(
                min(255, floor(sr / n / 16) * 16 + 8),
                min(255, floor(sg / n / 16) * 16 + 8),
                min(255, floor(sb / n / 16) * 16 + 8))
        end
        result[oy + 1] = table.concat(row)
        validQuota()
    end
    return result
end


local _hasSaveRT = false   -- set when the helper render target exists

local _quant = {}
for v = 0, 255 do _quant[v] = min(255, floor(v / 16) * 16 + 8) end

-- reads size x size pixels of a render target, one row per capture
local function readRows(rtName, size, reduceColors, progressCallback, p0, p1)
    local char = string.char
    local rowStrings = {}

    for y = 0, size - 1 do
        progressCallback(p0 + (p1 - p0) * y / size)
        render.selectRenderTarget(rtName)
        render.capturePixels()
        local row = {}
        if reduceColors then
            for x = 0, size - 1 do
                local c = render.readPixel(x, y)
                row[x + 1] = char(_quant[c.r], _quant[c.g], _quant[c.b])
            end
        else
            for x = 0, size - 1 do
                local c = render.readPixel(x, y)
                row[x + 1] = char(c.r, c.g, c.b)
            end
        end
        rowStrings[y + 1] = table.concat(row)
        validQuota()
    end
    return rowStrings
end

-- looks at a few rows: photos change color almost every pixel, drawings do not
local function looksLikePhoto()
    if not _hasSaveRT then return false end

    local changes = 0
    local samples = { 40, 170, 300, 430 }
    render.selectRenderTarget("myrendertarget")
    render.capturePixels()
    for _, y in ipairs(samples) do
        local pr, pg, pb
        for x = 0, _screenSize - 1 do
            local c = render.readPixel(x, y)
            if c.r ~= pr or c.g ~= pg or c.b ~= pb then
                changes = changes + 1
                pr, pg, pb = c.r, c.g, c.b
            end
        end
    end
    validQuota()
    return changes / #samples > PhotoRunThreshold
end

-- bilinear copy of the canvas into the top left corner of "saveRT"
local function drawShrunkCanvas(scale)
    render.selectRenderTarget("saveRT")
    render.clear(Color(0, 0, 0, 255))
    render.setFilterMin(2)
    render.setFilterMag(2)
    render.setColor(WHITE)
    render.setRenderTargetTexture("myrendertarget")
    render.drawTexturedRect(0, 0, 1024 / scale, 1024 / scale)
    filterImage()
    render.selectRenderTarget()
end


-- returns the compressed picture and the scale it was saved with (1 = full resolution)
function getCompressedScreenData(progressCallback)
    local char = string.char

    -- photo: shrink on the GPU, read only a quarter of the pixels
    if looksLikePhoto() then
        drawShrunkCanvas(2)
        local rows2 = readRows("saveRT", _screenSize / 2, true, progressCallback, 0.05, 0.7)
        local best = SCREEN_MAGIC3 .. char(2) .. encodeChunks(rows2, progressCallback, 0.7, 1)
        local scale = 2
        if #best > MaxSaveBytes then
            local rows4 = downscaleRows(rows2, 2, _screenSize / 2)
            best = SCREEN_MAGIC3 .. char(4) .. encodeChunks(rows4, progressCallback, 0.9, 1)
            scale = 4
        end
        progressCallback(1)
        return best, scale
    end

    -- drawing: full resolution, lossless
    local rowStrings = readRows("myrendertarget", _screenSize, false, progressCallback, 0, 0.7)
    local best = SCREEN_MAGIC .. encodeChunks(rowStrings, progressCallback, 0.7, 0.85)
    local scale = 1

    if #best > MaxSaveBytes then
        for _, sc in ipairs({ 2, 4 }) do
            local small = downscaleRows(rowStrings, sc)
            best = SCREEN_MAGIC3 .. char(sc) .. encodeChunks(small, progressCallback, 0.85, 1)
            scale = sc
            if #best <= MaxSaveBytes then break end
        end
    end

    progressCallback(1)
    return best, scale
end


-- returns the raw data of a saved picture and its scale (all formats)
local function decodeScreen(binary)
    local magic = string.sub(binary, 1, 4)
    local pos, scale
    if magic == SCREEN_MAGIC then
        pos, scale = 5, 1
    elseif magic == SCREEN_MAGIC3 then
        scale = string.byte(binary, 5)
        pos = 6
    else
        return bit.decompress(binary), 1
    end

    local parts = {}
    local total = #binary
    while pos + 3 <= total do
        local b1, b2, b3, b4 = string.byte(binary, pos, pos + 3)
        local len = ((b1 * 256 + b2) * 256 + b3) * 256 + b4
        pos = pos + 4
        local raw = bit.decompress(string.sub(binary, pos, pos + len - 1))
        if not raw then return nil end
        parts[#parts + 1] = raw
        pos = pos + len
        validQuota()
    end
    return table.concat(parts), scale
end


-- reads the canvas and sends it to the server with the given net message
local function uploadScreen(messageName, doneCallback)
    if st.saving then return false end
    st.saving = true
    st.progress = 0

    hook.add("renderoffscreen", "uploadScreen", function()
        local ok = true
        try(function()
            local compressedData, scale = getCompressedScreenData(function(value)
                st.progress = value
            end)
            if scale and scale > 1 then
                toast("Large picture: sent at 1/" .. scale .. " resolution")
            end
            net.start(messageName)
            net.writeData2(compressedData)
            net.send()
            --print(#compressedData)
            --print(#compressedData / 1024) -- KB
        end, function(err)
            ok = false
            toast("Save error: " .. errText(err))
        end)

        render.selectRenderTarget()
        st.saving = false
        hook.remove("renderoffscreen", "uploadScreen")
        if ok and doneCallback then doneCallback() end
    end)
    return true
end

-- bucket fill: reads the canvas lazily (re-capturing after every pause), draws the
-- spans immediately and sends them to the others
local function startFill(sx, sy)
    if st.filling or st.saving then return end
    st.filling = true
    pushRecent()
    local color = st.color

    hook.add("renderoffscreen", "fillTool", function()
        local S = _screenSize
        local needCapture = true

        local reads = 0
        local pause
        local function px(x, y)
            reads = reads + 1
            if reads % 128 == 0 then pause() end
            if needCapture then
                render.selectRenderTarget("myrendertarget")
                render.capturePixels()
                needCapture = false
            end
            return render.readPixel(x, y)
        end

        pause = function()
            while not checkQuota() do
                coroutine.yield()
                needCapture = true
            end
        end

        local t = px(sx, sy)
        local tr, tg, tb = t.r, t.g, t.b

        if not (tr == color.r and tg == color.g and tb == color.b) then
            local vis = {}
            local function match(x, y)
                if vis[y * S + x] then return false end
                local c = px(x, y)
                return c.r == tr and c.g == tg and c.b == tb
            end

            local spans = {}
            local function flush()
                if #spans == 0 then return end
                local op = { T = "F", C = color, Sp = spans }
                spans = {}
                sendOp(op)
            end

            local stack, sp = { sx, sy }, 2
            local n = 0
            while sp > 0 do
                local y = stack[sp]
                local x = stack[sp - 1]
                sp = sp - 2

                if match(x, y) then
                    local l, r = x, x
                    while l > 0 and match(l - 1, y) do l = l - 1 end
                    while r < S - 1 and match(r + 1, y) do r = r + 1 end
                    for xx = l, r do vis[y * S + xx] = true end

                    spans[#spans + 1] = y
                    spans[#spans + 1] = l
                    spans[#spans + 1] = r - l + 1

                    for dy = -1, 1, 2 do
                        local ny = y + dy
                        if ny >= 0 and ny < S then
                            local inRun = false
                            for xx = l, r do
                                if match(xx, ny) then
                                    if not inRun then
                                        inRun = true
                                        stack[sp + 1] = xx
                                        stack[sp + 2] = ny
                                        sp = sp + 2
                                    end
                                else
                                    inRun = false
                                end
                            end
                        end
                    end

                    n = n + 1
                    if #spans >= 300 then flush() end
                end
            end
            flush()
        end

        render.selectRenderTarget()
        st.filling = false
        hook.remove("renderoffscreen", "fillTool")
    end)
end

end

local _keyPressedData = {
    [MOUSE.MOUSE1] = { Pressed = false, IsFirst = false, IsReleased = false, IsMouse = true },
    [MOUSE.MOUSE2] = { Pressed = false, IsFirst = false, IsReleased = false, IsMouse = true },
    [MOUSE_MIDDLE] = { Pressed = false, IsFirst = false, IsReleased = false, IsMouse = true },
    [KEY.LALT] = { Pressed = false, IsFirst = false, IsReleased = false },
    [KEY.LSHIFT] = { Pressed = false, IsFirst = false, IsReleased = false },
    [KEY.E] = { Pressed = false, IsFirst = false, IsReleased = false },
}

-- "click" = left mouse button or the E key
local _click = { Pressed = false, IsFirst = false, IsReleased = false }

local function pollInput()
    for k, v in pairs(_keyPressedData) do
        local newPressed = v.IsMouse and input.isMouseDown(k) or input.isKeyDown(k)
        v.IsFirst = newPressed and not v.Pressed
        v.IsReleased = v.Pressed and not newPressed
        v.Pressed = newPressed
    end

    local m1, e = _keyPressedData[MOUSE.MOUSE1], _keyPressedData[KEY.E]
    local newClick = m1.Pressed or e.Pressed
    _click.IsFirst = newClick and not _click.Pressed
    _click.IsReleased = _click.Pressed and not newClick
    _click.Pressed = newClick

    return _keyPressedData, _click
end

local function toolbarHandler(cur, click)
    if not click.Pressed then st.active = nil end

    if st.active then
        if st.active.OnDrag then st.active:OnDrag(cur) end
        return
    end

    local hit = nil
    for _, w in ipairs(_widgets) do
        if (w.OnPress or w.OnDrag)
            and cur.X >= w.X and cur.X < w.X + w.W and cur.Y >= w.Y and cur.Y < w.Y + w.H
            and hasButtonRights(w) then
            hit = w
            break
        end
    end

    if hit ~= st.hover then
        st.hover = hit
        st.dirty = true
    end

    if hit and click.IsFirst then
        st.active = hit
        if hit.OnPress then hit:OnPress(cur) end
        if hit.OnDrag then hit:OnDrag(cur) end
        st.dirty = true
    end
end

        for i = 1, min(300, #_bufferNewData) do
            list[i] = table.remove(_bufferNewData, 1)
        end
        net.start("addNewClientData")
        net.writeTable(list)
        net.send(nil, false, true)
    end
end)

        for i, data in ipairs(list) do

                runOp(data, clientRect)
            end
        end
        render.selectRenderTarget()
        hook.remove("renderoffscreen", hookName)
    end)
end)


-- a whole picture (Draw From Dupe, image import)
net.receive("syncScreen", function()
    local binaryData = net.readData2()
    if not binaryData or binaryData == "" then return end
    local data, scale = decodeScreen(binaryData)
    if not data then return end
    scale = scale or 1
    local size = floor(_screenSize / scale)

    st.syncId = (st.syncId + 1) % 1000
    local hookName = "syncScreen_" .. st.syncId

    hook.add("renderoffscreen", hookName, function()
        local byte = string.byte
        local idx = 1

        for y = 0, size - 1 do
            render.selectRenderTarget("myrendertarget")
            filterImage()

            local x = 0
            while x < size do
                local r, g, b = byte(data, idx, idx + 2)
                if not b then break end

                local run = 1
                local j = idx + 3
                while x + run < size do
                    local r2, g2, b2 = byte(data, j, j + 2)
                    if r2 == r and g2 == g and b2 == b then
                        run = run + 1
                        j = j + 3
                    else
                        break
                    end
                end

                render.setRGBA(r, g, b, 255)
                render.drawRectFast(x * scale, y * scale, run * scale, scale)
                x = x + run
                idx = idx + run * 3
            end

            st.syncRow = y * scale
            validQuota()
        end

        st.syncRow = nil
        render.selectRenderTarget()
        hook.remove("renderoffscreen", hookName)
    end)
end)

---------- Image import (owner only, url comes from the wire inputs) ----------------

local function importImage(url)
    if st.importing or _p ~= _o then return end
    st.importing = true
    toast("Loading image...")

    local mat = material.create("UnlitGeneric")
    mat:setTextureURL("$basetexture", url, function(m, u, w, h, layout)
        if not w then
            st.importing = false
            toast("Image could not be loaded (check the url / permissions)")
            return
        end

        -- the image is stretched over the whole texture, we draw it back with the right aspect
        if layout then layout(0, 0, 1024, 1024) end
        local aspect = w / h
        local dw, dh = _screenSize, _screenSize
        if aspect > 1 then dh = floor(_screenSize / aspect) else dw = floor(_screenSize * aspect) end
        local ox, oy = floor((_screenSize - dw) / 2), floor((_screenSize - dh) / 2)

        timer.simple(0.6, function()
            hook.add("renderoffscreen", "importImage", function()
                render.selectRenderTarget("myrendertarget")
                render.setFilterMin(2)
                render.setFilterMag(2)
                render.setColor(CleanColor)
                render.drawRectFast(0, 0, _screenSize, _screenSize)
                render.setColor(WHITE)
                render.setMaterial(mat)
                render.drawTexturedRect(ox, oy, dw, dh)
                filterImage()

                render.selectRenderTarget()
                st.importing = false
                hook.remove("renderoffscreen", "importImage")
                toast("Image drawn, sending it to everybody...")
                uploadScreen("broadcastScreen")
            end)
        end)
    end)
end

local function cancelInteraction()
    st.prevPen = nil
    st.prevErase = nil
    st.shapeStart = nil
end


local function shapeEnd(sx, sy, x, y, kp)
    if not kp[KEY.LSHIFT].Pressed then return x, y end

    local dx, dy = x - sx, y - sy
    if st.tool == "line" then
        -- snap to horizontal / vertical / 45 degrees
        if abs(dx) > 2 * abs(dy) then return x, sy end
        if abs(dy) > 2 * abs(dx) then return sx, y end
    end
    local d = max(abs(dx), abs(dy))
    return sx + (dx < 0 and -d or d), sy + (dy < 0 and -d or d)
end


local function shapeBox(sx, sy, ex, ey)
    local x0, y0 = min(sx, ex), min(sy, ey)
    return x0, y0, abs(ex - sx) + 1, abs(ey - sy) + 1
end


local function commitShape(x, y, kp)
    local s = st.shapeStart
    if not s then return end
    local ex, ey = shapeEnd(s.x, s.y, x, y, kp)
    pushRecent()

    if st.tool == "line" then
        emit(lineOp(s.x, s.y, ex, ey, st.size, st.round, st.color, true))
    else
        local x0, y0, w, h = shapeBox(s.x, s.y, ex, ey)
        if st.tool == "rect" then
            if st.filled then
                emit({ T = "R", X = x0, Y = y0, W = w, H = h, C = st.color, S = true })
            else
                emit({ T = "B", X = x0, Y = y0, W = w, H = h, Sz = st.size, C = st.color, S = true })
            end
        else
            emit({ T = "E", X = x0, Y = y0, W = w, H = h, Sz = st.size, F = st.filled, Rd = st.round, C = st.color, S = true })
        end
    end
end


local function useTool(x, y, kp, click)
    local tool = st.tool

    if tool == "pen" then
        if click.Pressed then
            if click.IsFirst then
                pushRecent()
                if st.rainbowPen then advanceRainbow() end
                if kp[KEY.LSHIFT].Pressed and st.lastPoint then
                    emit(lineOp(st.lastPoint.x, st.lastPoint.y, x, y, st.size, st.round, st.color, true))
                else
                    emit(lineOp(x, y, x, y, st.size, st.round, st.color, true))
                end
                st.prevPen = { x = x, y = y }
            elseif st.prevPen and (x ~= st.prevPen.x or y ~= st.prevPen.y) then
                if st.rainbowPen then advanceRainbow() end
                emit(lineOp(st.prevPen.x, st.prevPen.y, x, y, st.size, st.round, st.color, false))
                st.prevPen = { x = x, y = y }
            end
            st.lastPoint = { x = x, y = y }
        else
            st.prevPen = nil
        end

    elseif tool == "line" or tool == "rect" or tool == "oval" then
        if click.IsFirst then st.shapeStart = { x = x, y = y } end
        if click.IsReleased and st.shapeStart then
            commitShape(x, y, kp)
            st.shapeStart = nil
        end

    elseif tool == "fill" then
        if click.IsFirst then startFill(x, y) end

    elseif tool == "pick" then
        if click.IsFirst then
            pickColorAt(x, y)
            st.tool = st.prevTool or "pen"
            st.dirty = true
        end

    end
end


local function renderMainHandler(curx, cury, kp, click)
    local x = math.clamp(floor(curx), 0, _screenSize - 1)
    local y = math.clamp(floor(cury), 0, _screenSize - 1)

    -- cog button
    if click.Pressed and curx < 20 and cury < 20 then
        if click.IsFirst then
            st.open = not st.open
            st.dirty = true
            st.active = nil
            st.hover = nil
            st.lockDraw = true
            cancelInteraction()
        end
        return
    end

    if not hasRights() then return end

    if st.open then
        toolbarHandler({ X = curx, Y = cury }, click)
        return
    end

    local m2 = kp[MOUSE.MOUSE2]

    if st.lockDraw then
        if not click.Pressed and not m2.Pressed then st.lockDraw = false end
        return
    end

    -- quick pick
    if kp[MOUSE_MIDDLE].IsFirst then
        pickColorAt(x, y)
        return
    end

    -- eraser
    if m2.Pressed then
        local sz = kp[KEY.LALT].Pressed and EraserBigSize or EraserSmallSize
        if st.prevErase then
            emit(lineOp(st.prevErase.x, st.prevErase.y, x, y, sz, true, CleanColor, false))
        else
            emit(lineOp(x, y, x, y, sz, true, CleanColor, true))
        end
        st.prevErase = { x = x, y = y }
        st.prevPen = nil
        st.shapeStart = nil
        return
    else
        st.prevErase = nil
    end

    useTool(x, y, kp, click)
end

local _toolNames = { pen = "Pen", line = "Line", rect = "Rect", oval = "Oval", fill = "Fill", pick = "Pick" }
local _mirrorNames = { [0] = "", "  [Mirror X]", "  [Mirror Y]", "  [Mirror XY]" }

local function drawShapePreview(curx, cury, kp)
    local s = st.shapeStart
    if not s then return end

    local x = math.clamp(floor(curx), 0, _screenSize - 1)
    local y = math.clamp(floor(cury), 0, _screenSize - 1)
    local ex, ey = shapeEnd(s.x, s.y, x, y, kp)

    local c = st.color
    render.setColor(Color(c.r, c.g, c.b, 200))

    if st.tool == "line" then
        render.drawLine(s.x + 0.5, s.y + 0.5, ex + 0.5, ey + 0.5)
    else
        local x0, y0, w, h = shapeBox(s.x, s.y, ex, ey)
        if st.tool == "rect" then
            render.drawRectOutline(x0, y0, w, h)
        else
            local rx, ry = w / 2, h / 2
            local n = 40
            local px, py
            for i = 0, n do
                local a = i / n * 2 * math.pi
                local qx = x0 + rx + rx * math.cos(a)
                local qy = y0 + ry + ry * math.sin(a)
                if px then render.drawLine(px, py, qx, qy) end
                px, py = qx, qy
            end
        end
    end
end


local function drawBrushRing(curx, cury, kp)
    local d, round
    if kp[MOUSE.MOUSE2].Pressed or kp[KEY.LALT].Pressed then
        d = kp[KEY.LALT].Pressed and EraserBigSize or EraserSmallSize
        round = true
    elseif st.tool == "pen" or st.tool == "line" or st.tool == "oval" then
        d, round = st.size, st.round
    elseif st.tool == "rect" then
        d, round = st.size, false
    else
        return
    end
    if d < 3 then return end

    render.setColor(Color(0, 0, 0, 200))
    if round then
        render.drawCircle(floor(curx) + 0.5, floor(cury) + 0.5, d / 2)
    else
        render.drawRectOutline(floor(curx) - floor(d / 2), floor(cury) - floor(d / 2), d, d)
    end
end


local function drawHud(curx, cury)
    render.setColor(Color(0, 0, 0, 140))
    render.drawRectFast(0, 494, 512, 18)
    render.setColor(WHITE)
    render.setFont(font2)
    local text = string.format("%s   %dpx   %s   (%d, %d)%s%s",
        _toolNames[st.tool] or st.tool, st.size, colorHex(st.color),
        floor(curx), floor(cury), _mirrorNames[st.mirror] or "", st.rainbowPen and "  [Rainbow]" or "")
    render.drawText(4, 496, text)
    render.setColor(st.color)
    render.drawRectFast(496, 497, 12, 12)
    render.setColor(WHITE)
    render.drawRectOutline(496, 497, 12, 12)
end

    if not st.looking or st.open or not hasRights() then return end
    if delta > 0 then
        st.size = math.clamp(max(st.size + 1, floor(st.size * 1.2 + 0.5)), 1, MaxSize)

        st.size = math.clamp(min(st.size - 1, floor(st.size / 1.2)), 1, MaxSize)
    end
    st.dirty = true
end)
