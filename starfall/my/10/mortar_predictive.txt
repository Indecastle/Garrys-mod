local function angDiff(a, b)
    return ((a - b + 180) % 360) - 180
end

-- Muzzle position and barrel direction. elev is the angle above the horizon.
local function getMuzzlePos(origin, yawDeg, elevDeg)
    local y, el = math.rad(yawDeg), math.rad(elevDeg)
    local dir = Vector(math.cos(el) * math.cos(y), math.cos(el) * math.sin(y), math.sin(el))
    return origin + Vector(0, 0, PLATE_H + PIVOT_H) + dir * MUZZLE_LEN, dir
end

-- Ballistics at a fixed angle: find the flight time t at which the shell
-- hits a (possibly moving) target. Returns dir, v0, t or nil.
local function solveBallistic(from, to, vel, pitchDeg, g)
    local th = math.rad(pitchDeg)
    local cosT, sinT = math.cos(th), math.sin(th)
    if cosT < 1e-4 then return nil end
    local tanT = sinT / cosT

    local function horiz(t)
        local dx = to.x - from.x + vel.x * t
        local dy = to.y - from.y + vel.y * t
        return dx, dy, math.sqrt(dx * dx + dy * dy)
    end

    local function f(t)
        local _, _, d = horiz(t)
        local dz = to.z - from.z + vel.z * t
        return d * tanT - 0.5 * g * t * t - dz
    end

    local lo = 0.01
    if f(lo) <= 0 then return nil end

    local hi
    local t = lo
    while t < 300 do
        t = t * 1.1
        if f(t) <= 0 then hi = t break end
        lo = t
    end
    if not hi then return nil end

    for _ = 1, 50 do
        local mid = (lo + hi) * 0.5
        if f(mid) > 0 then lo = mid else hi = mid end
    end

    local tf = (lo + hi) * 0.5
    local dx, dy, d = horiz(tf)
    if d < 1e-3 then return nil end

    local v0 = d / (tf * cosT)
    local dir = Vector(dx / d * cosT, dy / d * cosT, sinT)
    return dir, v0, tf
end

    local function rot(v, ang)
        local r = Vector(v.x, v.y, v.z)
        r:rotate(ang)
        return r
    end

    -- Telescopic leg: the outer tube hangs from the clamp, the inner one extends to the foot pad.
    local function placeLeg(outer, inner, top, foot)
        local d = top - foot
        local len = d:getLength()
        if len < 1 then return end
        local dir = d:getNormalized()
        local ang = d:getAngle() + Angle(90, 0, 0)

        local outerLen = math.min(len * 0.62, 34)
        local innerLen = len - outerLen + 10          -- 10 units of overlap inside the outer tube

        outer:setSize(Vector(5, 5, outerLen))
        outer:setPos(top - dir * (outerLen * 0.5))
        outer:setAngles(ang)

        inner:setSize(Vector(3, 3, innerLen))
        inner:setPos(foot + dir * (innerLen * 0.5))
        inner:setAngles(ang)
    end

    local function applyPose()
        local basePos = _e:getPos()
        local baseAng = Angle(0, baseYaw, 0)
        local turPos  = basePos + Vector(0, 0, PLATE_H)
        local turAng  = Angle(0, pose.yaw, 0)
        local pivot   = turPos + Vector(0, 0, PIVOT_H)
        local barAng  = Angle(90 - pose.elev, pose.yaw, 0)
        local barPos  = pivot - barAng:getUp() * (pose.recoil * RECOIL_VIS)

        local frames = {
            base   = { basePos, baseAng },
            turret = { turPos, turAng },
            barrel = { barPos, barAng },
        }

        for _, p in ipairs(PARTS) do
            local f = frames[p.frame]
            p.holo:setPos(f[1] + rot(p.lpos, f[2]))
            p.holo:setAngles(f[2] + p.lang)
        end

        -- bipod: leg tops on the clamp axle, bottoms in the foot pads
        local footL = turPos + rot(Vector(FOOT_X,  FOOT_Y, FOOT_Z), turAng)
        local footR = turPos + rot(Vector(FOOT_X, -FOOT_Y, FOOT_Z), turAng)
        local topL  = barPos + rot(Vector(0,  YOKE_W, YOKE_S), barAng)
        local topR  = barPos + rot(Vector(0, -YOKE_W, YOKE_S), barAng)
        placeLeg(legs[1].outer, legs[1].inner, topL, footL)
        placeLeg(legs[2].outer, legs[2].inner, topR, footR)

        -- spreader bar between the legs
        local pl = footL + (topL - footL) * SPREAD_K
        local pr = footR + (topR - footR) * SPREAD_K
        spreader:setPos((pl + pr) * 0.5)
        spreader:setAngles(turAng + ALONG_Y)

        -- sight: mounted on the tube, but its body always stays level and points along the azimuth
        local sightPos = barPos + rot(Vector(0, 15, SIGHT_S), barAng)
        local fwd  = turAng:getForward()
        local sAng = turAng + ALONG_X
        sightBody:setPos(sightPos);              sightBody:setAngles(sAng)
        sightEye:setPos(sightPos - fwd * 7.2);   sightEye:setAngles(sAng)
        sightLens:setPos(sightPos + fwd * 6.2);  sightLens:setAngles(sAng)
    end
    applyPose()

    hook.add("think", "mortarPose", function()
        if not alive then return end
        local dy = angDiff(state.yaw, pose.yaw)
        local de = state.pitch - pose.elev
        if math.abs(dy) < 0.05 and math.abs(de) < 0.05 and pose.recoil < 0.02 then return end

        pose.yaw  = pose.yaw + dy * 0.15
        pose.elev = pose.elev + de * 0.15
        pose.recoil = pose.recoil > 0.02 and pose.recoil * 0.94 or 0
        applyPose()
    end)

    local function sendExplosion(pos)
        net.start("explode")

    local function blast(pos, damage, radius)
        local done = false
        if hasPermission("game.blastDamage") then
            done = pcall(game.blastDamage, pos, radius, damage)
        end
        if done then return end
        if not prop.canSpawn() then return end

        local ent = prop.createSent(pos, Angle(), "gmod_wire_explosive", true, {
            damage = damage,
            radius = radius,
            removeafter = true,
            Model = "models/bull/gates/logic_nano.mdl",
        })
        ent._isSF = true
        ent:setSolid(false)
        ent:setNoDraw(true)
        ent:getWirelink()["Detonate"] = 1
    end

    local function hasShield(user)
        if user:hasGodMode()
                or (user.isBUILD ~= nil and user:isBUILD())
                or (user.isSafezoneSafe ~= nil and user:isSafezoneSafe()) then
            return true
        end
        local kyle = user:getNWVar("_Kyle_Buildmode")
        return kyle ~= nil and kyle
    end

    local function playExplodeSound(pos, holo)
        local path = "weapons/mortar/mortar_explode" .. math.random(1, 3) .. ".wav"
        if sound.emitSound then
            sound.emitSound(path, pos, 100, nil, 1, 0, nil, true)
        elseif isValid(holo) then
            holo:emitSound(path, 100, nil, 1, 0, nil, true)
        end
    end

    local function detonate(shell, withBlast)
        if shell._done then return end
        shell._done = true

        local pos = isValid(shell) and shell:getPos() or nil
        local trail = shell._trail
        if pos then
            if isValid(trail) then
                trail:setParent(nil)
                playExplodeSound(pos, trail)
                timer.simple(CFG.trailTime, function()
                    if isValid(trail) then trail:remove() end
                end)
            else
                playExplodeSound(pos, nil)
            end
            sendExplosion(pos)
            if withBlast then blast(pos, CFG.blastDamage, CFG.blastRadius) end
            shell:remove()
        end
    end

    local function damageShell(shell, amount)
        if not isValid(shell) or not shell._isBullet or shell._done then return end
        shell._health = math.max(0, shell._health - amount)
        if shell._health == 0 then
            detonate(shell, false)   -- a shot-down shell explodes without area damage
        end
    end

    local fireSound = sound.create(_e, "weapons/mortar/mortar_fire1.wav", true)
    fireSound:setSoundLevel(90)

    local function spawnShell(muzzle, dir, v0, t, ang)
        if not prop.canSpawn() then return false end

        local shell = prop.create(muzzle, ang, CFG.shellModel, false)
        local trail = hologram.create(muzzle, ang, MDL_CUBE, Vector(1))
        trail:setNoDraw(true)
        trail:setTrails(50, 0, CFG.trailTime, "trails/smoke", Color(255, 255, 255))
        trail:setParent(shell)

        shell._trail = trail
        shell._isBullet = true
        shell._health = CFG.shellHealth
        shell._done = false

        local phys = shell:getPhysicsObject()
        phys:setDragCoefficient(-10000)
        shell:setMass(0)
        shell:setFrozen(false)

        local vel = dir * v0
        phys:setVelocity(vel)
        -- set the velocity again: physics may reset it during the first ticks
        timer.simple(0.01, function()
            if not isValid(shell) then return end
            shell:setFrozen(false)
            shell:getPhysicsObject():setVelocity(vel)
            timer.simple(0.01, function()
                if not isValid(shell) then return end
                shell:setFrozen(false)
                shell:getPhysicsObject():setVelocity(vel)
            end)
        end)

        timer.simple(t + 0.07, function()
            if isValid(shell) then detonate(shell, true) end
        end)
        shell:addCollisionListener(function()
            if isValid(shell) then detonate(shell, true) end
        end)
        if t > 1.2 then
            timer.simple(t - 1.2, function()
                if not isValid(shell) or shell._done then return end
                shell:emitSound("weapons/mortar/mortar_shell_incomming1.wav", 100, nil, 1, nil, nil, true)
            end)
        end
        return true
    end

    -- The server picks the impact point itself: a trace straight down from the camera.
    local function computeShot(cam, pitch)
        local chipPos = _e:getPos()
        local tr = trace.line(cam, cam - Vector(0, 0, 100000), nil, MASK.ALL, nil, false)
        local target = tr.HitPos
        local flat = Vector(target.x - chipPos.x, target.y - chipPos.y, 0)
        local dist = flat:getLength()
        if dist < CFG.minRadius or dist > CFG.maxRadius then return nil end

        local yaw = flat:getAngle().y
        local muzzle = getMuzzlePos(chipPos, yaw, pitch)
        local dir, v0, t = solveBallistic(muzzle, target, Vector(), pitch, GRAVITY)
        if not dir then return nil end
        return dir, v0, t, muzzle, yaw
    end

    ----------------------------------------------------------------
    -- network
    ----------------------------------------------------------------
    local function sanitizeCam(cam)
        local base = _e:getPos()
        local flat = Vector(cam.x - base.x, cam.y - base.y, 0)
        local maxR = CFG.maxRadius
        if flat:getLength() > maxR then
            flat = flat:getNormalized() * maxR
        end
        local z = math.clamp(cam.z, base.z + 200, base.z + 30000)
        return Vector(base.x + flat.x, base.y + flat.y, z)
    end

    local viewHolo = hologram.create(_e:getPos() + Vector(0, 0, CFG.startHeight), Angle(), MDL_CUBE, Vector(1))
    viewHolo:setNoDraw(true)

    net.receive("SetData", function(_, ply)
        if not alive or not isValid(vehicle) then return end
        if ply ~= vehicle:getDriver() then return end

        local cam = sanitizeCam(net.readVector())
        local pitch = math.clamp(net.readUInt(8), CFG.minPitch, CFG.maxPitch)

        state.cam, state.pitch = cam, pitch
        local flat = Vector(cam.x - _e:getPos().x, cam.y - _e:getPos().y, 0)
        if flat:getLength() > 1 then state.yaw = flat:getAngle().y end

        pcall(function()
            viewHolo:setPos(CFG.usePVS and cam or Vector(0, 0, -1000000))
            ply:setViewEntity(viewHolo)
        end)
    end)

    net.receive("tryFire", function(_, ply)
        if not alive or reloading or not isValid(vehicle) then return end
        if ply ~= vehicle:getDriver() then return end

        local cam = sanitizeCam(net.readVector())
        local pitch = math.clamp(net.readUInt(8), CFG.minPitch, CFG.maxPitch)

        local dir, v0, t, muzzle, yaw = computeShot(cam, pitch)
        if not dir then return end

        -- snap the barrel to the target pose so the shell leaves from the muzzle
        state.cam, state.pitch, state.yaw = cam, pitch, yaw
        pose.yaw, pose.elev = yaw, pitch
        pose.recoil = 9
        applyPose()

        local ang = Angle(90 - pitch, yaw, 0)
        if not spawnShell(muzzle, dir, v0, t, ang) then return end

        reloading = true
        fireSound:stop()
        fireSound:play()

        flash:setPos(muzzle)
        flash:setNoDraw(false)
        timer.simple(0.07, function()
            if isValid(flash) then flash:setNoDraw(true) end
        end)

        net.start("fire")
        net.writeBool(true)
        net.writeFloat(timer.curtime() + CFG.reloadTime)
        net.writeVector(muzzle)
        net.send()

        timer.simple(CFG.reloadTime, function()
            reloading = false
            net.start("fire")
            net.writeBool(false)
            net.send()
        end)
    end)

    timer.create("scanPlayers", 1, 0, function()
        if not isValid(vehicle) then return end
        local drv = vehicle:getDriver()
        if not isValid(drv) then return end

        local list = find.allPlayers(function(p)
            return p ~= drv and not hasShield(p)
        end)
        net.start("players")
        net.writeTable(list)
        net.send(drv)
    end)

    ----------------------------------------------------------------
    -- death and damage to the mortar
    ----------------------------------------------------------------
    local function killingScene()
        alive = false
        local pos = _e:getPos() + Vector(0, 0, 40)
        sendExplosion(pos)
        blast(pos, CFG.blastDamage, CFG.blastRadius)
        for _, p in ipairs(PARTS) do
            if isValid(p.holo) then p.holo:remove() end
        end
        for _, h in ipairs(DYNAMIC) do
            if isValid(h) then h:remove() end
        end
        if isValid(flash) then flash:remove() end
        timer.simple(10, function()
            if isValid(_e) then _e:remove() end
        end)
    end

    hook.add("EntityTakeDamage", "mortarDamage", function(target, attacker, inflictor, amount)
        if not alive then return end

        -- hit markers for the driver
        local drv = isValid(vehicle) and vehicle:getDriver() or nil
        if isValid(drv)
                and (attacker == _o or (isValid(inflictor) and inflictor._isSF))
                and isValid(target) and target:isPlayer() and target:isAlive()
                and not hasShield(target) then
            net.start("hitmark")
            net.writeVector(target:getPos())
            net.writeInt(math.round(amount, 0), 32)
            net.send(drv)
        end

        if isValid(target) and target._isBullet then
            damageShell(target, amount)
        end

        if target == _e and health > 0 then
            health = math.max(0, health - amount)
            setDamageTint(true)
            if health <= 0 then
                timer.remove("mortarTint")
                killingScene()
            else
                timer.create("mortarTint", 0.5, 1, function() setDamageTint(false) end)
            end
        end
    end)

    hook.add("PlayerEnteredVehicle", "mortarEnter", function(ply, veh)
        if not alive then return end
        if CFG.pvpOnly and isValid(ply) and veh == vehicle and hasShield(ply) then
            vehicle:ejectDriver()
        end
    end)

    hook.add("ClientInitialized", "mortarInit", function(ply)
        if not alive then return end
        net.start("init")
        net.writeEntity(vehicle)
        net.send(ply)
    end)

    return
end

local emitter

function spawnParticle(pos, mat, sizeStart, sizeEnd, speed, life)
    if not emitter then
        if particle.particleEmittersLeft() < 1 then return end
        emitter = particle.create(Vector(), false)
    end
    if emitter:getParticlesLeft() < 1 then return end

    local dir = Vector(math.rand(-1, 1), math.rand(-1, 1), math.rand(0, 1)):getNormalized()
    local part = emitter:add(mat, pos + dir * math.random(0, 50), sizeStart, sizeEnd, 0, 0, 50, 0, life)
    if not part then return end
    part:setCollide(true)
    part:setBounce(0)
    part:setVelocity(dir * speed)
    part:setAirResistance(100)
end

    for _ = 1, 5 do spawnParticle(pos, mSmoke, 100, 1000, 500, 10) end
    for _ = 1, 5 do spawnParticle(pos, mFlame, 100, 1000, 500, 10) end
end)

local aimRate = 6
local aimSnap = false -- flag: apply result immediately

local function computeAim()
    if not isActive() then return end

    local chipPos = _e:getPos()
    local tr = trace.line(camPos, camPos - Vector(0, 0, 100000), nil, MASK.ALL, nil, false)
    local target = tr.HitPos

    --       (.  )
    if aim.tTarget and (aim.tTarget - target):getLength() < 8 then
        target = aim.tTarget
    end

    local flat = Vector(target.x - chipPos.x, target.y - chipPos.y, 0)
    local dist = flat:getLength()

    aim.tTarget    = target
    aim.distTarget = dist
    aim.rawReason  = nil
    aim.rawOk      = false

    if dist < CFG.minRadius then aim.rawReason = "TOO CLOSE"   return end
    if dist > CFG.maxRadius then aim.rawReason = "OUT OF RANGE" return end

    local muzzle = getMuzzlePos(chipPos, flat:getAngle().y, pitch)
    local dir, v0, t = solveBallistic(muzzle, target, Vector(), pitch, GRAVITY)
    if not dir then aim.rawReason = "NO SOLUTION" return end

    aim.rawOk = true
    aim.rawReason = "READY"
    aim.rawMuzzle, aim.rawDir, aim.rawV0, aim.rawT = muzzle, dir, v0, t
end

--   raw*  aim ( )
local function snapAim()
    if not aim.rawOk then
        aim.ok, aim.reason = false, aim.rawReason or "NO DATA"
        --     ,    
        -- aim.dir, aim.muzzle, aim.t, aim.v0 = nil, nil, nil, nil
        return
    end
    aim.target = Vector(aim.tTarget.x, aim.tTarget.y, aim.tTarget.z)
    aim.dist   = aim.distTarget
    aim.muzzle = Vector(aim.rawMuzzle.x, aim.rawMuzzle.y, aim.rawMuzzle.z)
    aim.dir    = Vector(aim.rawDir.x, aim.rawDir.y, aim.rawDir.z)
    aim.v0     = aim.rawV0
    aim.t      = aim.rawT
    aim.ok     = true
    aim.reason = "READY"
end

timer.create("aimSolver", 0.05, 0, function()
    computeAim()
    if aimSnap then
        snapAim()
        aimSnap = false
    end
end)

----------------------------------------------------------------
-- smooth interpolation helpers
----------------------------------------------------------------
local function damp(cur, goal, rate, dt)
    -- FPS-independent exponential approximation
    return cur + (goal - cur) * (1 - math.exp(-rate * dt))
end

local function dampVec(cur, goal, rate, dt)
    if not cur then return Vector(goal.x, goal.y, goal.z) end
    return Vector(
        damp(cur.x, goal.x, rate, dt),
        damp(cur.y, goal.y, rate, dt),
        damp(cur.z, goal.z, rate, dt)
    )
end

-- called every frame while the HUD is active
hook.add("think", "aimSmooth", function()
    if not isActive() then return end

    local now = timer.curtime()
    local d = math.min(now - (aim._last or now), 0.1)
    aim._last = now
    if d <= 0 then return end

    --   snap       
    if aimSnap then return end

    if aim.tTarget then
        aim.target = dampVec(aim.target, aim.tTarget, aimRate, d)
        aim.dist   = damp(aim.dist or aim.distTarget or 0, aim.distTarget or 0, aimRate, d)
    end

    if aim.rawOk then
        aim.t      = damp(aim.t or aim.rawT, aim.rawT, aimRate, d)
        aim.v0     = damp(aim.v0 or aim.rawV0, aim.rawV0, aimRate, d)
        aim.dir    = dampVec(aim.dir, aim.rawDir, aimRate, d)
        aim.muzzle = dampVec(aim.muzzle, aim.rawMuzzle, aimRate, d)
        local l = aim.dir:getLength()
        if l > 0 then aim.dir = aim.dir / l end
        aim.ok, aim.reason = true, "READY"
    else
        aim.ok, aim.reason = false, aim.rawReason or aim.reason or "NO DATA"
    end
end)

local function drawTrajectory()
    if not aim.ok then return end

    render.setColor(Color(255, 220, 80, 170))
    local prev
    local steps = 28
    for i = 0, steps do
        local tt = aim.t * i / steps
        local p = aim.muzzle + aim.dir * aim.v0 * tt - Vector(0, 0, 0.5 * GRAVITY * tt * tt)
        local s = p:toScreen()
        if s.visible and prev then render.drawLine(prev.x, prev.y, s.x, s.y) end
        prev = s.visible and s or nil
    end

    -- blast radius ring around the impact point
    render.setColor(Color(255, 90, 60, 200))
    local last
    local seg = 24
    for i = 0, seg do
        local a = (i / seg) * math.pi * 2
        local p = aim.target + Vector(math.cos(a) * CFG.blastRadius, math.sin(a) * CFG.blastRadius, 5)
        local s = p:toScreen()
        if s.visible and last then render.drawLine(last.x, last.y, s.x, s.y) end
        last = s.visible and s or nil
    end
end

end

hook.add("mouseWheeled", "mortarWheel", function(delta)
    if not isActive() then return end

    local newPitch = math.clamp(pitch + CFG.pitchStep * delta, CFG.minPitch, CFG.maxPitch)
    if newPitch == pitch then return end
    pitch = newPitch

    -- reset the "catch-up" and recalculate immediately
    aimSnap = true
    computeAim()
    -- apply immediately (do not wait for 0.1 s timer)
    if aim.rawOk then
        snapAim()
    end
end)

local function isValidPos(from, to)
    local tr = trace.hull(from, to, Vector(-50, -50, -100), Vector(50, 50, 300), nil, MASK.VISIBLE, nil, false)
    local base = _e:getPos()
    local flat = Vector(to.x - base.x, to.y - base.y, 0)
    return not tr.Hit and flat:getLength() < CFG.maxRadius
end

hook.add("calcview", "mortarView", function(_, _, _, znear, zfar)
    local now = timer.curtime()
    dt = math.min(now - lastTime, 0.1)
    lastTime = now

    if not isActive() then return end

    local v = Vector()
    if keys[KEY.W] then v = v + Vector(1, 0, 0) end
    if keys[KEY.S] then v = v - Vector(1, 0, 0) end
    if keys[KEY.A] then v = v + Vector(0, 1, 0) end
    if keys[KEY.D] then v = v - Vector(0, 1, 0) end
    if keys[KEY.SPACE] then v = v + Vector(0, 0, 1) end
    if keys[KEY.ALT] then v = v - Vector(0, 0, 1) end

    if v:getLength() ~= 0 then
        local speed = SPEED
        if keys[KEY.CTRL] then speed = SPEED_SLOW
        elseif keys[KEY.SHIFT] then speed = SPEED_FAST end

        -- the higher the camera, the faster it moves
        local height = math.max(camPos.z - _e:getPos().z, 1000)
        local nextPos = camPos + v:getNormalized() * speed * dt * 70 * (height / 1000)
        if isValidPos(camPos, nextPos) then camPos = nextPos end
    end

    return {
        origin = camPos,
        angles = Angle(90, 0, 0),
        fov = 120,
        znear = znear,
        zfar = zfar,
        drawviewer = true,
    }
end)
