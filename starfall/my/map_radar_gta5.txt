// ---- map bounds: center of the map + size covering the whole map ----
local function boundsOk(a, b)
    if not (a and b) then return false end
    local dx, dy = b.x - a.x, b.y - a.y
    if dx < 512 or dy < 512 or dx > 66000 or dy > 66000 then return false end
    local c = chip():getPos()
    return c.x >= a.x and c.x <= b.x and c.y >= a.y and c.y <= b.y
end

local function noEnts() return false end

local function rayBounds()
    local o = chip():getPos() + Vector(0, 0, 50)
    local top = trace.line(o, o + Vector(0, 0, 32000), noEnts, MASK.SOLID_BRUSHONLY).HitPos.z
    local h = Vector(o.x, o.y, top - 48)
    local function reach(d) return trace.line(h, h + d * 33000, noEnts, MASK.SOLID_BRUSHONLY).HitPos end
    local xp, xn, yp, yn = reach(Vector(1, 0, 0)), reach(Vector(-1, 0, 0)), reach(Vector(0, 1, 0)), reach(Vector(0, -1, 0))
    local bot = trace.line(o, o - Vector(0, 0, 32000), noEnts, MASK.SOLID_BRUSHONLY).HitPos.z
    return Vector(xn.x, yn.y, bot - 512), Vector(xp.x, yp.y, top + 64)
end

if AutoMap then
    local mn, mx = BoundsMin, BoundsMax
    if not boundsOk(mn, mx) then
        local s, a, b = pcall(function() return game.getWorld():getModelBounds() end)
        if s and boundsOk(a, b) then
            mn, mx = a, b
        else
            s, a, b = pcall(rayBounds)
            if s and boundsOk(a, b) then
                mn, mx = a, b
            else
                mn, mx = Vector(-16384, -16384, -16384), Vector(16384, 16384, 16384)
            end
        end
    end
    local zTop = CamHeight or (mx.z + 32)
    local zBot = mn.z - 32
    SizeMap = math.max(mx.x - mn.x, mx.y - mn.y)
    InitPos = Vector((mn.x + mx.x) / 2, (mn.y + mx.y) / 2, zTop)
    TraceDist = math.min(RenderDistanceLimit, zTop - zBot + 64)
else
    InitPos = chip():getPos() + Vector(0,0, CamHeight and (CamHeight - chip():getPos().z) or Z)
end

function myIsInWorld(pos)
    return not trace.trace(pos, pos, nil, nil, 20).HitWorld
end


local tc_buffer = {}//to reduce lag
function texturedCircle(pos,size,segments)
    local id = pos.x.."_"..pos.y.."_"..size.y.."_"..size.y.."_"..segments
    if not tc_buffer[id] then
        local circle = {}
        for i=0, segments, 1 do
            local sin, cos = math.sin(math.rad((i/segments)*-360)), math.cos(math.rad((i/segments)*-360))
            local data = {
                x = (sin*size.x)+pos.x,
                y = (cos*size.y)+pos.y,
                u = (sin*0.5)+0.5,
                v = (cos*0.5)+0.5
            }
            table.insert(circle,data)
        end
        tc_buffer[id] = circle
    end
    render.drawPoly(tc_buffer[id])
end


local tr_buffer = {} // to reduce lag
function texturedRectangle(pos, size)
    local id = pos.x .. "_" .. pos.y .. "_" .. size.x .. "_" .. size.y
    
    if not tr_buffer[id] then
        local rect = {
            { x = pos.x - size.x, y = pos.y - size.y, u = 0, v = 0 },  
            { x = pos.x + size.x, y = pos.y - size.y, u = 1, v = 0 }, 
            { x = pos.x + size.x, y = pos.y + size.y, u = 1, v = 1 },  
            { x = pos.x - size.x, y = pos.y + size.y, u = 0, v = 1 }   
        }
        tr_buffer[id] = rect
    end
    
    render.drawPoly(tr_buffer[id])
end


function getArrow(poly, curX, curY, scale)
    local copyMouse = table.copy(poly)
    
    for i=1, #copyMouse do
        copyMouse[i].x = copyMouse[i].x*scale + curX
        copyMouse[i].y = copyMouse[i].y*scale + curY
    end
    return copyMouse
end


function drawArrow(ply, curX, curY, scale)
    render.setRenderTargetTexture()
    local poly
    local plyZ = ply:getPos().z
    local OwnerZ = OO:getPos().z
    if plyZ > OwnerZ+LevelDiff then
        poly = polyUp
    elseif plyZ < OwnerZ-LevelDiff then
        poly = polyDown
    else
        poly = polyRect
    end
    copyMouse = getArrow(poly, curX, curY-20, scale)

    render.drawPoly(copyMouse)
end

    local pos, _ = worldToLocal(InitPos, Angle(), OO:getPos(), eyeAngles)
    return  -pos.y / SizeMap * _zoomSmooth,
            -pos.x / SizeMap * _zoomSmooth
end

function get_coro()
    return coroutine.wrap(function()
            local I_ = 0
            local time1 = timer.curtime()
            for y=0, Res_ do
                for x=0, Res_ do
                    I_ = I_ + 1
                    if I_ == Performance then
                        I_ = 0
                        coroutine.yield()
                    end
                    
                    local tr
                    local startPos = InitPos + Vector(HalfSizeMap_ - x*MapDxy, HalfSizeMap_ - y*MapDxy)
                    local endPos = startPos + Dir * TraceDist
                    local dist = -1
                    for i=0, 2 do
                        tr = trace.trace(startPos, endPos, nil, 24737, nil, false)
                        if myIsInWorld(tr.HitPos) then
                            dist = startPos:getDistance(tr.HitPos)
                            break
                        else
                            startPos = tr.HitPos + Dir*1
                        end
                    end
                    
                    local color = Color(0,0,0,0)
                    if dist ~= -1 then
                        if bit.band(tr.SurfaceFlags, 8) ~= 0 then
                            color = Color(0,200,200)
                        else
                            color = render.traceSurfaceColor(tr.HitNormal + tr.HitPos, tr.HitPos - tr.HitNormal)
                        end
                        
                        render.setColor(color)
                        render.drawRectFast(x*Detail,Res_*Detail - y*Detail, Detail, Detail)
                    end
                    
                end
                //coroutine.yield()
            end
            //print(timer.curtime() - time1)
            return true
        end)
end

function createTiltedRadarPoly(centerX, centerY, width, height, tilt, rot)
    local halfWidth = width / 2
    local tiltedHeight = height * tilt
       
    local rotRad = math.rad(rot)
    local cosRot = math.cos(rotRad)
    local sinRot = math.sin(rotRad)
    
    local basePoints = {
        { x = -halfWidth * tilt, y = -height/2 },
        { x = halfWidth * tilt, y = -height/2 },
        { x = halfWidth, y = height/2 },
        { x = -halfWidth, y = height/2 }
    }
    
    --      
    local rotatedPoly = {}
    for i, point in ipairs(basePoints) do
        -- x' = x*cos() - y*sin()
        -- y' = x*sin() + y*cos()
        local rotatedX = point.x * cosRot - point.y * sinRot
        local rotatedY = point.x * sinRot + point.y * cosRot
        
        table.insert(rotatedPoly, {
            x = centerX + rotatedX,
            y = centerY + rotatedY,
            u = (i == 1 or i == 4) and 0 or 1,
            v = (i == 1 or i == 2) and 0 or 1
        })
    end
    
    return rotatedPoly
end

// Map through render.renderView: orthographic camera from above, same area (SizeMap / auto map bounds) and
// same orientation as the trace scan (texture: right = -X, down = +Y), so the rest of the code is untouched
function captureMapRenderView()
    local hs = HalfSizeMap_

    render.selectRenderTarget("myrendertarget")
    render.clear(Color(0,0,0,255), true)

    local ok, err = pcall(render.renderView, {
        origin = Vector(InitPos.x, InitPos.y, InitPos.z),
        angles = Angle(90, -90, 0),
        x = 0, y = 0, w = 1024, h = 1024,
        aspectratio = 1,
        ortho = {left = -hs, right = hs, top = -hs, bottom = hs},
        znear = 1,
        zfar = TraceDist,
        drawviewmodel = false,
        drawmonitors = false,
        drawviewer = false,
        dopostprocess = false,
        bloomtone = false,
    })

    if ok then
        // renderView leaves garbage alpha -> force alpha 255
        render.overrideBlend(true, BLEND.ZERO, BLEND.ONE, BLENDFUNC.ADD, BLEND.ONE, BLEND.ZERO, BLENDFUNC.ADD)
        render.setRGBA(255, 255, 255, 255)
        render.drawRect(0, 0, 1024, 1024)
        render.overrideBlend(false)
    else
        render.clear(Color(0,0,0,0), true)
    end

    render.selectRenderTarget()
    return ok, err
end


// Map through trace (fallback)
function traceRender()
    if canRenderView() then return end // renderView will draw the map (see "renderscene")

    render.selectRenderTarget("myrendertarget")
    if _ifFirstTime then
        _ifFirstTime = false
        render.clear(Color(0,0,0,0))
    end

    while math.max(quotaAverage(), quotaUsed()) < quotaMax() * Performance_Cpu_Limit do
        if paint() then
            hook.remove("renderoffscreen","")
            return
            //paint = get_coro()
        end
    end
end


function startTrace()
    paint = get_coro()
    _ifFirstTime = true
    hook.add("renderoffscreen","",traceRender)
end


function initHooks()
    startTrace()

    hook.add("renderscene","",function()
        if not rvCapture or render.isInRenderView() or not canRenderView() then return end
        rvCapture = false

        local ok, err = captureMapRenderView()
        if ok then
            hook.remove("renderoffscreen","") // map is ready, trace is not needed anymore
        else
            rvFailed = true
            print("[MapRadar] renderView failed, using trace: " .. tostring(err))
            startTrace()
        end
    end)

    hook.add("drawhud","",function()    
        if Mode != 0 then   
            _zoomSmooth = _zoomSmooth + (Zoom_ - _zoomSmooth)/10
            local size = _zoomSmooth / 2
            
            local offsetX, offsetY = getOffset()

            render.selectRenderTarget("myrendertarget2")
            render.setRenderTargetTexture("myrendertarget")
            render.clear(Color(0, 0, 0, 50))
            
            if Filtering then
                render.setFilterMin(1)
                render.setFilterMag(1)
            end
            
            local cc1 = math.clamp(CenterMapCoef, 0, 1)
            local cx = 512
            local cy = 1024 - 512 * (cc1)
            
            --   
            local eyeAngles = OO:getEyeAngles()
            local rot = eyeAngles.y * -1 - 90
            local radRot = math.rad(-rot)
            
            local cosR = math.cos(radRot)
            local sinR = math.sin(radRot)
            
            --local grad = 40 * math.sin(timer.systime() % 360 * 1)
            local tilt = math.cos(math.rad(Titl)) 
            
            local function getCorner(sx, sy)
                local rx = sx * cosR - sy * sinR
                local ry = sx * sinR + sy * cosR
                
                local mapX = rx + offsetX
                local mapY = ry + offsetY
                  
                return Vector(cx + mapX, cy + mapY * tilt, 0)
            end
            
            render.draw3DQuad(
                getCorner(-size, -size),
                getCorner(size, -size),  
                getCorner(size, size),   
                getCorner(-size, size),  
                Color(255, 255, 255, 255)
            )

            renderPlayers(cx, cy, tilt)
            renderCenterMouse(cx, cy)

            render.selectRenderTarget("myrendertarget3")
            render.setRenderTargetTexture("myrendertarget2")
            render.clear(Color(0,0,0,0))
            
            local MapSizeX = MapSize * MapScale
            local MapSizeY = MapSize * MapScale
            
            local mapPosX = IsRightPos and (_width - MapSizeX - MapPosX) or (MapPosX)
            local mapPosY = IsBottomPos and (_height - MapSizeY/2 - MapPosY) or (MapPosY)
            
            render.setColor(Color(255,255,255,255))
            texturedRectangle(Vector(512, 0), Vector(512, 512))
            
            render.selectRenderTarget()
            render.setRenderTargetTexture("myrendertarget3")

            render.drawTexturedRect(mapPosX, mapPosY, MapSizeX, MapSizeY)
            
            --render.setColor(Color())
            --render.drawCircle(mapPosX + MapSizeX/2, mapPosY + MapSizeY/4 + (cy-768)/256*MapSizeY/4, 3)
            --render.setMaterial(cursorTex)
            --render.setColor(Color(100,100,100))
            --render.drawTexturedRect(mapPosX + MapSizeX/2, mapPosY + MapSizeY/4 + (cy-768)/256*MapSizeY/4, 20, 25)
            
            renderBar(mapPosX, mapPosY, MapSizeX, MapSizeY)
        end
        
        if Mode == 2 and (isOwner or !OnlyOwner) then
            drawConectingPlayers()
        end
    end)    
end


function hudConnecting(isCon)
    if !ON then
        ON = true
        initHooks()
    end
    Mode = isCon and 1 or 0
    net.start("hud_connecting")
    net.writeBool(isCon)
    net.send()
end

        else

        end
    end

    end

        end
